{%- comment -%} One JSON array entry per HTML page in this language: title, url, the text of its h2/h3 headings, and the first 300 characters of its text, so a 600-page site keeps its index well under a megabyte. Pages with `search: false`, pages with `layout: none`, and non-HTML output (a URL whose last segment ends in a file extension other than .html/.htm, such as llms.txt or a data feed) stay out. {%- endcomment -%} {%- assign sl_ix_lang = page.lang | default: site.lang | default: 'en' -%} {%- assign sl_ix_all = site.pages | concat: site.documents -%} {%- assign sl_ix_n = 0 -%} [ {%- for item in sl_ix_all -%} {%- assign sl_ix_item_lang = item.lang | default: site.lang | default: 'en' -%} {%- assign sl_ix_file = item.url | default: '' | split: '/' | last -%} {%- assign sl_ix_ext = sl_ix_file | split: '.' | last -%} {%- if sl_ix_item_lang != sl_ix_lang or item.search == false or item.layout == 'search-index' or item.layout == 'none' or item.title == nil or item.url == nil or item.url == '' -%}{%- continue -%}{%- endif -%} {%- if sl_ix_file contains '.' and sl_ix_ext != 'html' and sl_ix_ext != 'htm' -%}{%- continue -%}{%- endif -%} {%- assign sl_ix_html = item.content | markdownify -%} {%- capture sl_ix_heads -%} {%- assign sl_ix_parts = sl_ix_html | split: '' | last | strip_html | strip -%} {%- if sl_ix_head != '' %}{{ sl_ix_head }} {% endif -%} {%- endif -%} {%- endif -%} {%- endfor -%} {%- endcapture -%} {%- if sl_ix_n > 0 -%},{%- endif -%} {%- assign sl_ix_n = sl_ix_n | plus: 1 %} {"title":{{ item.title | jsonify }},"url":{{ item.url | relative_url | jsonify }},"headings":{{ sl_ix_heads | normalize_whitespace | truncate: 300 | jsonify }},"text":{{ sl_ix_html | strip_html | normalize_whitespace | truncate: 300 | jsonify }}} {%- endfor %} ]
